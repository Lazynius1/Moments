rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    // =====================================================================
    // HELPER FUNCTIONS
    // =====================================================================

    // ✅ ACTUALIZADA: Función helper para verificar si un usuario puede ver contenido
    function canViewContent(viewerId, ownerId) {
      let ownerDoc = get(/databases/$(database)/documents/users/$(ownerId));
      return viewerId == ownerId ||
             (ownerDoc.data.get('isActive', true) && // ✅ NUEVO: Verificar que la cuenta esté activa
              (!ownerDoc.data.isPrivate ||
               exists(/databases/$(database)/documents/users/$(viewerId)/following/$(ownerId))));
    }

    // ✅ NUEVA: Función helper más permisiva para operaciones de feed
    function canReadForFeed(viewerId, ownerId) {
      return viewerId == ownerId ||
             exists(/databases/$(database)/documents/users/$(ownerId)) &&
             get(/databases/$(database)/documents/users/$(ownerId)).data.get('isActive', true);
    }

    // ✅ CORREGIDA: Función helper para verificar audiencia personalizada
    function canViewCustomContent(viewerId, ownerId, contentType, contentId) {
      let audienceDocId = contentType + '_' + contentId;
      return exists(/databases/$(database)/documents/users/$(ownerId)/customAudiences/$(audienceDocId)) &&
             viewerId in get(/databases/$(database)/documents/users/$(ownerId)/customAudiences/$(audienceDocId)).data.allowedUsers;
    }

    // ✅ FUNCIÓN HELPER NUEVA: Verificar si un usuario está en una lista personalizada
    function isUserInCustomList(viewerId, ownerId, customListId) {
      return customListId != '' &&
             exists(/databases/$(database)/documents/users/$(ownerId)/customAudienceLists/$(customListId)) &&
             viewerId in get(/databases/$(database)/documents/users/$(ownerId)/customAudienceLists/$(customListId)).data.members;
    }

    // ✅ ACTUALIZADA: Función helper para verificar audiencia completa con cuentas activas
        function canViewContentWithAudience(viewerId, ownerId, audience, contentType, contentId, customListId) {
          let ownerDoc = get(/databases/$(database)/documents/users/$(ownerId));
          return viewerId == ownerId ||
                 (ownerDoc.data.get('isActive', true) &&
                  canViewContent(viewerId, ownerId) &&
                   (audience == 'everyone' ||
                    (audience == 'connections' &&
                     exists(/databases/$(database)/documents/users/$(viewerId)/following/$(ownerId)) &&
                     exists(/databases/$(database)/documents/users/$(ownerId)/following/$(viewerId))) ||
                    (audience == 'bestFriends' &&
                     viewerId in ownerDoc.data.bestFriends) ||
                    (audience == 'custom' &&
                     canViewCustomContent(viewerId, ownerId, contentType, contentId)) ||
                    (audience == 'customList' &&
                     isUserInCustomList(viewerId, ownerId, customListId)) ||
                    (audience == 'onlyMe' && viewerId == ownerId)
                   )
                 );
        }

    // ✅ NUEVA: Función helper para verificar si una cuenta puede ser seguida
    function canFollowUser(followerId, targetUserId) {
      let followerUserDoc = get(/databases/$(database)/documents/users/$(followerId));
      let targetUserDoc = get(/databases/$(database)/documents/users/$(targetUserId));
      return followerUserDoc.data.get('isActive', true) && targetUserDoc.data.get('isActive', true); // Ambos usuarios deben estar activos
    }

    // ✅ NUEVA: Función para verificar si un usuario puede participar en conversaciones
    function canParticipateInConversations(userId) {
      return get(/databases/$(database)/documents/users/$(userId)).data.get('isActive', true);
    }
    
        // ✅ NUEVA: Función para validar memoria categorizada de Nova
        function isValidNovaMemory(memoryData) {
          return memoryData.keys().hasAll(['id', 'userId', 'facts', 'lastUpdated', 'createdAt']) &&
                 memoryData.id is string &&
                 memoryData.userId is string &&
                 memoryData.facts is list &&
                 memoryData.facts.size() <= 20 && // Máximo 20 hechos categorizados
                 memoryData.lastUpdated is timestamp &&
                 memoryData.createdAt is timestamp &&
                 // Validar que cada hecho sea un objeto válido
                 isValidCategorizedFactsList(memoryData.facts);
        }

        // ✅ NUEVA: Función para validar lista de hechos categorizados
        function isValidCategorizedFactsList(factsList) {
          return factsList.size() == 0 || // Lista vacía es válida
                 (factsList.size() > 0 && isValidNovaFact(factsList[0])); // Solo validar el primer elemento para simplificar
        }

        // ✅ NUEVA: Función para validar un hecho categorizado individual
        function isValidNovaFact(fact) {
          return fact.keys().hasAll(['id', 'content', 'type', 'timestamp', 'importance']) &&
                 fact.id is string &&
                 fact.content is string &&
                 fact.content.size() > 0 &&
                 fact.content.size() <= 300 && // Máximo 300 caracteres por hecho
                 fact.type is string &&
                 fact.type in ['preference', 'personal', 'professional', 'interest', 'general'] &&
                 fact.timestamp is timestamp &&
                 fact.importance is int &&
                 fact.importance >= 1 &&
                 fact.importance <= 5;
        }

        // ✅ Prevenir spam de actualizaciones de memoria
        function canUpdateNovaMemory(userId) {
          return !exists(/databases/$(database)/documents/users/$(userId)/novaMemory/memory) ||
                 request.time > resource.data.lastUpdated + duration.value(30, 's');
        }

        // ✅ Validar tamaño total de datos de memoria
        function isMemorySizeValid(memoryData) {
          let totalSize = memoryData.facts.size() * 150; // ✅ Debe ser *
          return totalSize <= 5000; // Máximo 5KB de datos de memoria
        }
        
    
    // =====================================================================
    // USER COLLECTIONS
    // =====================================================================
        match /users/{userId} {
        // Regla de lectura sin cambios
        allow read: if request.auth != null && (
            request.auth.uid == userId ||
            resource.data.get('isActive', true) ||
            (request.auth != null && exists(/databases/$(database)/documents/users/$(request.auth.uid)))
            );
    
            // ✅ REGLA DE CREACIÓN CORREGIDA
            allow create: if request.auth != null
                && request.auth.uid == userId
                && request.resource.data.id == userId
                // ✅ CAMBIO CRÍTICO: Eliminar la validación estricta del email
                && request.resource.data.email is string
                && request.resource.data.email.size() > 0
                // Validaciones básicas del modelo
                && request.resource.data.username is string
                && request.resource.data.username.size() > 0
                && request.resource.data.interests is list
                && request.resource.data.isPlusSubscriber is bool
                && request.resource.data.blockedUsers is list
                && request.resource.data.isPrivate is bool
                && request.resource.data.showMutualConnections is bool
                && request.resource.data.showFollowing is bool
                && request.resource.data.showAdmirers is bool
                && request.resource.data.bestFriends is list
                && request.resource.data.get('isActive', true) == true  // ✅ Asegurar que sea true
                && request.resource.data.get('showBadge', true) is bool
                && request.resource.data.get('showPlusBadge', true) is bool
                && request.resource.data.get('isVerified', false) is bool
                && request.resource.data.get('onlineStatus', 'offline') is string
                && request.resource.data.get('isOnline', false) is bool
                // Campos opcionales
                && (!("ownedBadges" in request.resource.data) || request.resource.data.ownedBadges is list)
                && (!("plusSubscription" in request.resource.data) || request.resource.data.plusSubscription is map)
                && (!("profileImagePath" in request.resource.data) || request.resource.data.profileImagePath is string)
                && (!("bio" in request.resource.data) || request.resource.data.bio is string)
                && (!("activeHoursStart" in request.resource.data) || request.resource.data.activeHoursStart is string)
                && (!("activeHoursEnd" in request.resource.data) || request.resource.data.activeHoursEnd is string)
                && (!("notificationPreferences" in request.resource.data) || request.resource.data.notificationPreferences is map)
                && (!("primaryBadgeId" in request.resource.data) || request.resource.data.primaryBadgeId is string)
                && (!("selectedProfileTheme" in request.resource.data) || request.resource.data.selectedProfileTheme is string)
                && (!("lastSeen" in request.resource.data) || request.resource.data.lastSeen is timestamp)
                    && (!("encryptionPublicKeyBase64" in request.resource.data) || request.resource.data.encryptionPublicKeyBase64 is string)
                // No permitir campos de desactivación en creación
                && !("deactivatedAt" in request.resource.data)
                && !("deactivatedBy" in request.resource.data)
                && !("reactivatedAt" in request.resource.data);

          // Permitir actualización de campos de desactivación/reactivación y estado en línea
          allow update: if request.auth != null && request.auth.uid == userId && (
            // Actualizaciones normales (cualquier campo excepto los de gestión de cuenta)
            !request.resource.data.diff(resource.data).affectedKeys().hasAny(['isActive', 'deactivatedAt', 'deactivatedBy', 'reactivatedAt']) ||
            // O actualizaciones específicas de gestión de cuenta
            (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['isActive', 'deactivatedAt', 'deactivatedBy']) &&
             request.resource.data.isActive == false &&
             request.resource.data.deactivatedAt is timestamp &&
             request.resource.data.deactivatedBy is string) ||
            (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['isActive', 'reactivatedAt']) &&
             request.resource.data.isActive == true &&
             request.resource.data.reactivatedAt is timestamp) ||
            (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['isActive', 'reactivatedAt', 'deactivatedAt', 'deactivatedBy']) &&
             request.resource.data.isActive == true &&
             request.resource.data.reactivatedAt is timestamp) ||
            // ✅ NUEVO: Regla para actualización de estado en línea
            (request.resource.data.diff(resource.data).affectedKeys().hasAny(['onlineStatus', 'lastSeen', 'isOnline']) &&
             (!("onlineStatus" in request.resource.data) || request.resource.data.onlineStatus is string) &&
             (!("lastSeen" in request.resource.data) || request.resource.data.lastSeen is timestamp) &&
             (!("isOnline" in request.resource.data) || request.resource.data.isOnline is bool))
          );

          allow delete: if request.auth != null && request.auth.uid == userId;

            // 🧠 SISTEMA DE MEMORIA PERSONALIZADA DE NOVA
            match /novaMemory/{memoryId} {
                allow read: if request.auth != null && request.auth.uid == userId;

                // Crear memoria con validaciones completas
                allow create: if request.auth != null
                              && request.auth.uid == userId
                              && get(/databases/$(database)/documents/users/$(userId)).data.get('isActive', true)
                              && isValidNovaMemory(request.resource.data)
                              && isMemorySizeValid(request.resource.data);

                // Actualizar memoria con protección anti-spam
                allow update: if request.auth != null
                              && request.auth.uid == userId
                              && get(/databases/$(database)/documents/users/$(userId)).data.get('isActive', true)
                              && isValidNovaMemory(request.resource.data)
                              && isMemorySizeValid(request.resource.data)
                              && canUpdateNovaMemory(userId);

                allow delete: if request.auth != null && request.auth.uid == userId;
            }
            
            // ✅ NUEVA: Regla para dispositivos del usuario
            match /devices/{deviceId} {
              allow read, write: if request.auth != null && 
                request.auth.uid == userId &&
                get(/databases/$(database)/documents/users/$(userId)).data.get('isActive', true);
            }

      // SISTEMA DE SEGUIMIENTO
      match /following/{followedUserId} {
        // ✅ CORREGIDA: Regla más permisiva para lectura de following
        allow read: if request.auth != null && (
          request.auth.uid == userId || // El propietario ve todos sus seguimientos
          (canViewContent(request.auth.uid, userId) && // ✅ Puedes ver el perfil del usuario
           get(/databases/$(database)/documents/users/$(userId)).data.showFollowing) // ✅ Y permite mostrar following
        );
        
        // ✅ REGLA DE PRODUCCIÓN: Seguir cuentas activas con validaciones
        allow create: if request.auth != null &&
                      request.auth.uid == userId &&
                      canFollowUser(userId, followedUserId) &&
                      // Verificar que no esté siguiendo ya
                      !exists(/databases/$(database)/documents/users/$(userId)/following/$(followedUserId));
        
        allow delete: if request.auth != null && request.auth.uid == userId;
        allow update: if request.auth != null && request.auth.uid == userId;
      }

      match /followers/{followerId} {
        // ✅ CORREGIDA: Regla más permisiva para lectura de followers
        allow read: if request.auth != null && (
          request.auth.uid == userId || // El propietario ve todos sus seguidores
          (canViewContent(request.auth.uid, userId) && // ✅ Puedes ver el perfil del usuario
           get(/databases/$(database)/documents/users/$(userId)).data.showAdmirers) // ✅ Y permite mostrar seguidores
        );
        
        // ✅ REGLA DE PRODUCCIÓN: Crear followers con validaciones de seguridad
        allow create: if request.auth != null && (
          request.auth.uid == followerId || // El follower puede crear su propio documento
          request.auth.uid == userId // El propietario del perfil puede crear documentos de followers
        ) &&
        // Verificar que no exista ya el follower
        !exists(/databases/$(database)/documents/users/$(userId)/followers/$(followerId));
        
        allow delete: if request.auth != null && (request.auth.uid == followerId || request.auth.uid == userId);
        allow update: if request.auth != null && (request.auth.uid == followerId || request.auth.uid == userId);
      }

      // SISTEMA DE SOLICITUDES DE SEGUIMIENTO
      match /sentFollowRequests/{requestId} {
        allow read: if request.auth != null && request.auth.uid == userId;
        // ✅ CORREGIDA: Solo permitir enviar solicitudes a cuentas activas
        allow create: if request.auth != null &&
                      request.auth.uid == userId &&
                      request.resource.data.recipientId is string &&
                      canFollowUser(userId, request.resource.data.recipientId);
        // ✅ CORREGIDA: Permitir que tanto el remitente como el destinatario actualicen
        // (necesario para el sistema de aceptación/rechazo de solicitudes)
        allow update: if request.auth != null && (
          request.auth.uid == userId || // El remitente puede actualizar
          request.auth.uid == resource.data.recipientId // El destinatario puede actualizar
        );
        // ✅ CORREGIDA: Permitir que tanto el remitente como el destinatario eliminen
        // (necesario para el sistema de limpieza de solicitudes)
        allow delete: if request.auth != null && (
          request.auth.uid == userId || // El remitente puede eliminar
          request.auth.uid == resource.data.recipientId // El destinatario puede eliminar
        );
      }

      match /receivedFollowRequests/{requestId} {
        allow read: if request.auth != null && request.auth.uid == userId;
        // ✅ CORREGIDA: Permitir que cualquier usuario autenticado cree solicitudes
        // (necesario para el sistema de solicitudes de seguimiento)
        allow create: if request.auth != null &&
                      request.resource.data.recipientId == userId &&
                      request.resource.data.senderId == request.auth.uid &&
                      canFollowUser(request.auth.uid, userId);
        // ✅ CORREGIDA: Permitir que tanto el remitente como el destinatario actualicen
        // (necesario para el sistema de aceptación/rechazo de solicitudes)
        allow update: if request.auth != null && (
          request.auth.uid == userId || // El destinatario puede actualizar
          request.auth.uid == resource.data.senderId // El remitente puede actualizar
        );
        allow delete: if request.auth != null && request.auth.uid == userId;
      }

      // ✅ ACTUALIZADO: Reglas para listas de audiencia personalizada
      match /customAudienceLists/{listId} {
        // ✅ PERMITIR LECTURA LIMITADA: Solo para verificar membresía en content visibility
        allow read: if request.auth != null && (
          // El propietario puede leer todas sus listas
          request.auth.uid == userId ||
          // Otros usuarios pueden leer SOLO para verificar si están en la lista
          // (esto es necesario para las security rules de contenido)
          request.auth.uid != null
        );

        allow create: if request.auth != null
                      && request.auth.uid == userId
                      && request.resource.data.keys().hasAll(['name', 'members', 'createdAt'])
                      && request.resource.data.name is string
                      && request.resource.data.name.size() > 0
                      && request.resource.data.name.size() <= 50
                      && request.resource.data.members is list
                      && request.resource.data.members.size() <= 100
                      && request.resource.data.createdAt is timestamp
                      && (!("description" in request.resource.data) || request.resource.data.description is string)
                      && (!("color" in request.resource.data) || request.resource.data.color is string)
                      && (!("icon" in request.resource.data) || request.resource.data.icon is string)
                      && (!("updatedAt" in request.resource.data) || request.resource.data.updatedAt is timestamp);

        allow update: if request.auth != null
                      && request.auth.uid == userId
                      && request.resource.data.name is string
                      && request.resource.data.name.size() > 0
                      && request.resource.data.name.size() <= 50
                      && request.resource.data.members is list
                      && request.resource.data.members.size() <= 100
                      && (!("description" in request.resource.data) || request.resource.data.description is string)
                      && (!("color" in request.resource.data) || request.resource.data.color is string)
                      && (!("icon" in request.resource.data) || request.resource.data.icon is string)
                      && (!("updatedAt" in request.resource.data) || request.resource.data.updatedAt is timestamp);

        allow delete: if request.auth != null && request.auth.uid == userId;
      }

      // Reglas para audiencias personalizadas de contenido específico
      match /customAudiences/{audienceId} {
        allow read: if request.auth != null && request.auth.uid == userId;

        allow create: if request.auth != null
                      && request.auth.uid == userId
                      && request.resource.data.keys().hasAll(['contentType', 'allowedUsers', 'createdAt'])
                      && request.resource.data.contentType in ['moment', 'story']
                      && request.resource.data.allowedUsers is list
                      && request.resource.data.allowedUsers.size() <= 100
                      && request.resource.data.createdAt is timestamp;

        allow update: if request.auth != null
                      && request.auth.uid == userId
                      && request.resource.data.contentType == resource.data.contentType
                      && request.resource.data.allowedUsers is list
                      && request.resource.data.allowedUsers.size() <= 100;

        allow delete: if request.auth != null && request.auth.uid == userId;
      }

      // ✅ ACTUALIZADO: Reglas mejoradas para momentos con configuración avanzada
      match /moments/{momentId} {
        allow read: if request.auth != null &&
                      canViewContentWithAudience(
                        request.auth.uid,
                        userId,
                        resource.data.get('audience', 'everyone'),
                        'moment',
                        momentId,
                        resource.data.get('customListId', '')
                      );

        allow create: if request.auth != null
                      && request.auth.uid == userId
                      && get(/databases/$(database)/documents/users/$(userId)).data.get('isActive', true)
                      && request.resource.data.authorId == userId
                      && (!("audience" in request.resource.data) ||
                          request.resource.data.audience in ['everyone', 'connections', 'bestFriends', 'custom', 'customList', 'onlyMe'])
                      && (!("customViewers" in request.resource.data) ||
                          request.resource.data.customViewers is list)
                      && (!("customListId" in request.resource.data) ||
                          request.resource.data.customListId is string)
                      && (!("customListName" in request.resource.data) ||
                          request.resource.data.customListName is string)
                      // ✅ NUEVAS VALIDACIONES: Campos de configuración avanzada
                      && (!("disableComments" in request.resource.data) ||
                          request.resource.data.disableComments is bool)
                      && (!("hideLikeCounts" in request.resource.data) ||
                          request.resource.data.hideLikeCounts is bool)
                      && (!("allowSharing" in request.resource.data) ||
                          request.resource.data.allowSharing is bool)
                      // ✅ NUEVA VALIDACIÓN: Campo de coordenadas de ubicación
                      && (!("locationCoordinate" in request.resource.data) ||
                          (request.resource.data.locationCoordinate is map &&
                           request.resource.data.locationCoordinate.latitude is number &&
                           request.resource.data.locationCoordinate.longitude is number));

        allow update: if request.auth != null && (
          request.auth.uid == userId ||
          (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['commentCount', 'lastActivity']) &&
           canViewContentWithAudience(request.auth.uid, userId, resource.data.get('audience', 'everyone'), 'moment', momentId, resource.data.get('customListId', ''))) ||
          (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['reactions']) &&
           canViewContentWithAudience(request.auth.uid, userId, resource.data.get('audience', 'everyone'), 'moment', momentId, resource.data.get('customListId', '')))
        );

        allow delete: if request.auth != null && request.auth.uid == userId;

        // ✅ ACTUALIZADO: Comentarios con validación de configuración avanzada
        match /comments/{commentId} {
          allow read: if request.auth != null &&
                         canViewContentWithAudience(
                           request.auth.uid,
                           userId,
                           get(/databases/$(database)/documents/users/$(userId)/moments/$(momentId)).data.get('audience', 'everyone'),
                           'moment',
                           momentId,
                           get(/databases/$(database)/documents/users/$(userId)/moments/$(momentId)).data.get('customListId', '')
                         );

          // ✅ NUEVO: Solo permitir crear comentarios si NO están deshabilitados
          allow create: if request.auth != null &&
                          request.resource.data.authorId == request.auth.uid &&
                          get(/databases/$(database)/documents/users/$(request.auth.uid)).data.get('isActive', true) &&
                          // ✅ VALIDAR: Comentarios habilitados en el momento
                          get(/databases/$(database)/documents/users/$(userId)/moments/$(momentId)).data.get('disableComments', false) == false &&
                           canViewContentWithAudience(
                             request.auth.uid,
                             userId,
                             get(/databases/$(database)/documents/users/$(userId)/moments/$(momentId)).data.get('audience', 'everyone'),
                             'moment',
                             momentId,
                             get(/databases/$(database)/documents/users/$(userId)/moments/$(momentId)).data.get('customListId', '')
                           );

          allow update: if request.auth != null && (
            request.auth.uid == resource.data.authorId ||
            (canViewContentWithAudience(
               request.auth.uid,
               userId,
               get(/databases/$(database)/documents/users/$(userId)/moments/$(momentId)).data.get('audience', 'everyone'),
               'moment',
               momentId,
               get(/databases/$(database)/documents/users/$(userId)/moments/$(momentId)).data.get('customListId', '')
             ) &&
             (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['reactions']) ||
              request.resource.data.diff(resource.data).affectedKeys().hasOnly(['reactions', 'metadata']) ||
              request.resource.data.diff(resource.data).affectedKeys().hasOnly(['metadata'])))
          );

          allow delete: if request.auth != null &&
                          (request.auth.uid == userId || request.auth.uid == resource.data.authorId);
        }

        // ✅ ACTUALIZADO: Reacciones con validación de configuración avanzada
        match /reactions/{reactionId} {
          allow read: if request.auth != null &&
                         canViewContentWithAudience(
                           request.auth.uid,
                           userId,
                           get(/databases/$(database)/documents/users/$(userId)/moments/$(momentId)).data.get('audience', 'everyone'),
                           'moment',
                           momentId,
                           get(/databases/$(database)/documents/users/$(userId)/moments/$(momentId)).data.get('customListId', '')
                         );

          // ✅ NUEVO: Solo permitir reacciones si NO están ocultos los likes
          allow create: if request.auth != null
                        && request.resource.data.userId == request.auth.uid
                        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.get('isActive', true)
                        // ✅ VALIDAR: Likes habilitados en el momento
                        && get(/databases/$(database)/documents/users/$(userId)/moments/$(momentId)).data.get('hideLikeCounts', false) == false
                        && canViewContentWithAudience(
                              request.auth.uid,
                              userId,
                              get(/databases/$(database)/documents/users/$(userId)/moments/$(momentId)).data.get('audience', 'everyone'),
                              'moment',
                              momentId,
                              get(/databases/$(database)/documents/users/$(userId)/moments/$(momentId)).data.get('customListId', '')
                            )
                        && request.resource.data.keys().hasAll(['userId', 'reactionType', 'timestamp'])
                        && request.resource.data.userId is string
                        && request.resource.data.reactionType is string
                        && request.resource.data.reactionType in ['vibe', 'fire', 'real', 'mood', 'glow', 'feel', 'love', 'wow', 'laugh', 'cry', 'respect', 'power', 'genius', 'creative', 'chill', 'hype']
                        && request.resource.data.timestamp is timestamp;

          allow update: if request.auth != null && request.auth.uid == resource.data.userId;

          allow delete: if request.auth != null && (
            request.auth.uid == userId ||
            request.auth.uid == resource.data.userId
          );
        }

        // ✅ ACTUALIZADO: Likes legacy con validación de configuración avanzada
        match /likes/{likeId} {
          allow read: if request.auth != null &&
                         canViewContentWithAudience(
                           request.auth.uid,
                           userId,
                           get(/databases/$(database)/documents/users/$(userId)/moments/$(momentId)).data.get('audience', 'everyone'),

    function isValidMessage(message) {
      return message.keys().hasAll(['id', 'conversationId', 'senderId', 'type', 'status', 'isRead', 'isDeleted', 'isViewed']) &&
             message.id is string &&
             message.conversationId is string &&
             message.senderId is string &&
             message.type in ['text', 'image', 'video', 'audio', 'location', 'file', 'gif', 'ephemeral', 'sharedMoment', 'viewOnceImage', 'viewOnceVideo'] &&
             message.status in ['sending', 'sent', 'delivered', 'read'] &&
             message.isRead is bool &&
             message.isDeleted is bool &&
             message.isViewed is bool &&
             (!("isViewOnce" in message) || message.isViewOnce is bool) &&
             (!('viewedBy' in message) || message.viewedBy is list) &&
             (!('statusUpdatedAt' in message) || message.statusUpdatedAt is timestamp);
    }

    function onlyStatusFieldsUpdated() {
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly(['isRead','status','isViewed','statusUpdatedAt']);

      return request.resource.data.diff(resource.data).affectedKeys().hasOnly(['isViewed','viewedBy']);

    function isAutomaticCleanup() {
      return resource.data.type == 'ephemeral' &&
             resource.data.expirationDate != null &&
             resource.data.expirationDate < request.time &&
             request.resource.data.isDeleted == true &&
             request.resource.data.deletedAt != null;
    }

    function isViewOnceCleanup() {
      return resource.data.get('isViewOnce', false) == true &&
             resource.data.type in ['viewOnceImage', 'viewOnceVideo'] &&
             request.resource.data.isDeleted == true &&
             request.resource.data.deletedAt != null;
